CODES = ["size", "extension", "mime", "magic", "active_content", "parses"]


@pytest.fixture()
def fresh_store(tmp_path):
    get_settings().object_dir = str(tmp_path / "objects")
    store.cache_clear()
    yield tmp_path / "objects"
    store.cache_clear()


def test_precheck_reports_every_check_and_the_scan_without_storing(api, fresh_store):
    a = api().login(ABC, INS)
    r = a.post("/api/documents/precheck", files={"file": ("fdr.pdf", pdf(), "application/pdf")})
    assert r.status_code == 200, r.text
    body = r.json()
    assert [c["code"] for c in body["checks"]] == CODES and all(c["ok"] for c in body["checks"])
    assert body["scan"]["status"] == "clean" and body["accepted"] is True and len(body["sha256"]) == 64
    assert not [p for p in fresh_store.rglob("*") if p.is_file()]  # nothing stored

    png = a.post("/api/documents/precheck", files={"file": ("scan.png", b"\x89PNG\r\n\x1a\n" + b"0" * 200, "image/png")}).json()
    oks = {c["code"]: c["ok"] for c in png["checks"]}
    assert oks["size"] is True and oks["extension"] is False and oks["mime"] is None and png["accepted"] is False
    assert png["scan"]["status"] == "not_run"


def test_precheck_flags_malware_and_audits_it(api, fresh_store):
    a = api().login(ABC, INS)
    body = a.post("/api/documents/precheck", files={"file": ("x.pdf", pdf(trailer=b"\n" + EICAR + b"\n"), "application/pdf")}).json()

    with SessionLocal() as db:

        assert ev and json.loads(ev[-1].data)["stage"] == "precheck"

def test_upload_keeps_content_checks_and_sha256_recheck_detects_tampering(api, fresh_store):
    a = api().login(ABC, INS)
    r = _upload(a, "SBI-CHK-0001", [("fdr.pdf", pdf(), "application/pdf"), ("lien.pdf", pdf("lien"), "application/pdf")])
    assert r.status_code == 200, r.text
    fid, doc = r.json()["fdr"]["id"], r.json()["documents"][0]
    assert [c["code"] for c in doc["content_checks"]] == CODES and all(c["ok"] for c in doc["content_checks"])
    assert doc["integrity"]["status"] is None

    ok = a.post(f"/api/fdrs/{fid}/documents/{doc['id']}/verify").json()
    assert ok["result"] == "match" and ok["actual"] == ok["expected"] == doc["sha256"]
    assert ok["document"]["integrity"]["status"] == "match"

    with SessionLocal() as db:   # replace the stored ciphertext with garbage
        key = db.get(m.FdrDocument, doc["id"]).object_key
    store().put(key, b"\x00" * 64)
    bad = a.post(f"/api/fdrs/{fid}/documents/{doc['id']}/verify").json()
    assert bad["result"].startswith("unreadable") and bad["actual"] is None
    with SessionLocal() as db:
        types = [e.type for e in db.scalars(select(m.PortalEvent).order_by(m.PortalEvent.seq))]
        assert "document_integrity_checked" in types and types[-1] == "document_integrity_failure"

"""Start the backend (uvicorn :8000) and the frontend (vite :5173) for the browser checks, on a disposable SQLite database.

Usage: .venv/bin/python tools/ui-checks/serve.py [--reset] [--prod] [--no-vite]

  --reset    delete the UI-check database first and run `scripts/seed.py --reset` (migrate to head + seed)
  --prod     TL_ENV=prod: no /api/sim/*, no mock OIDC, no dev secrets file (keys come from this process's environment)
  --no-vite  backend only

Real local services: ClamAV (:3310) scans every upload; SeaweedFS (tl-baseline-s3, :8334) holds the encrypted documents.
Credentials for SeaweedFS are read from the container's environment into the child process only; nothing is written to disk.
The database and audit checkpoint live under sih2/.ui-check (gitignored), never in the developer's database.
Demo password for every seeded user: TL_DEMO_PASSWORD (default UiCheck-Demo-2026, the value tools/ui-checks/lib.mjs uses).
"""
from __future__ import annotations

import base64
import json
import os
import secrets
import signal
import subprocess
import sys
from pathlib import Path

ROOT = Path(__file__).resolve().parents[2]
WORK = Path(os.environ.get("TL_UI_WORK", ROOT / ".ui-check"))
PY = str(ROOT / ".venv/bin/python")


def container_env(name: str) -> dict[str, str]:
    obj = json.loads(subprocess.check_output(["docker", "inspect", name]))[0]
    return dict(item.split("=", 1) for item in obj["Config"]["Env"] if "=" in item)


def main() -> None:
    reset, prod, no_vite = "--reset" in sys.argv, "--prod" in sys.argv, "--no-vite" in sys.argv
    WORK.mkdir(parents=True, exist_ok=True)
    s3 = container_env("tl-baseline-s3")
    env = {**os.environ,
           "TL_DATABASE_URL": f"sqlite:///{WORK / 'ui.db'}",
           "TL_CHECKPOINT_PATH": str(WORK / "checkpoints.jsonl"),
           "TL_DEMO_PASSWORD": os.environ.get("TL_DEMO_PASSWORD", "UiCheck-Demo-2026"),
           "TL_SCANNER": "clamd", "TL_CLAMD_HOST": "127.0.0.1", "TL_CLAMD_PORT": "3310",
           "TL_OBJECT_STORE": "s3", "TL_S3_ENDPOINT": "127.0.0.1:8334", "TL_S3_BUCKET": "tl-integration-tests",   # SeaweedFS here is capped at 8 volumes; reuse the bucket the tests use
           "TL_S3_ACCESS_KEY": s3["S3_ACCESS_KEY"], "TL_S3_SECRET_KEY": s3["S3_SECRET_KEY"],
           "TL_COOKIE_SECURE": "false"}
    if prod:
        env["TL_ENV"] = "prod"
        keys = WORK / "prod-keys.json"   # stable across restarts of the prod check; mode 600, outside the repository
        if not keys.exists():
            keys.write_text(json.dumps({"TL_SECRET_KEY": secrets.token_urlsafe(48), "TL_SIM_BANK_SEED": secrets.token_hex(32),
                                        "TL_MASTER_KEY": base64.b64encode(secrets.token_bytes(32)).decode(),
                                        "TL_DATA_KEY": base64.urlsafe_b64encode(secrets.token_bytes(32)).decode(),
                                        "TL_PROVIDER_WEBHOOK_SECRET": secrets.token_urlsafe(32)}))
            keys.chmod(0o600)
        env.update(json.loads(keys.read_text()))
    if reset:
        (WORK / "ui.db").unlink(missing_ok=True)
        (WORK / "checkpoints.jsonl").unlink(missing_ok=True)
        subprocess.run([PY, "scripts/seed.py", "--reset"], cwd=ROOT / "backend", env=env, check=True)
    procs = [subprocess.Popen([PY, "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1", "--port", "8000"], cwd=ROOT / "backend", env=env)]
    if not no_vite:
        procs.append(subprocess.Popen(["npx", "vite", "--host", "127.0.0.1", "--port", "5173", "--strictPort"], cwd=ROOT / "frontend"))

    def stop(*_):
        for p in procs:
            p.terminate()
    signal.signal(signal.SIGTERM, stop)
    signal.signal(signal.SIGINT, stop)
    for p in procs:
        p.wait()


if __name__ == "__main__":
    main()
